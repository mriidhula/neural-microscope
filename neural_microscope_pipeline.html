import torch, torch.nn as nn class SparseAutoencoder(nn.Module): """Maps 2048-d polysemantic residual → 8192-d monosemantic features.""" def __init__(self, d_model=2048, d_hidden=8192, l1_coeff=3e-4): super().__init__() self.W_enc = nn.Linear(d_model, d_hidden, bias=True) self.W_dec = nn.Linear(d_hidden, d_model, bias=True) self.l1_coeff = l1_coeff def forward(self, x): # Encode: dense → sparse features features = torch.nn.functional.relu(self.W_enc(x)) reconstructed = self.W_dec(features) # Loss: reconstruction + L1 sparsity penalty loss_recon = (reconstructed - x).pow(2).mean() loss_l1 = self.l1_coeff * features.abs().mean() return features, loss_recon + loss_l1 # Semantic labeling via Ollama def label_feature(feature_idx: int, top_activating_texts: list[str]) -> str: prompt = f"These texts maximally activate Feature #{feature_idx}:\n" prompt += "\n".join(top_activating_texts[:5]) prompt += "\n\nIn 3-5 words, what concept does this feature encode?" # → Calls Ollama locally, returns e.g. "Historical Apollo Moon Landings" return call_ollama(prompt)
